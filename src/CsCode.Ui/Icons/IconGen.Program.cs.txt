using System.Text;
using System.Text.Json;

// Scratch generator: dumps Codicon path data (100-unit box, y in [-100,0]) from the IconPacks package into a C# catalog.
var dllPath = Directory.GetFiles(AppContext.BaseDirectory, "IconPacks.Avalonia.Codicons.dll").Single();
var text = Encoding.UTF8.GetString(File.ReadAllBytes(dllPath));
var start = text.IndexOf("{\"None\":", StringComparison.Ordinal);
var depth = 0;
var end = start;
for (; end < text.Length; end++)
{
    if (text[end] == '{') depth++;
    else if (text[end] == '}' && --depth == 0) break;
}
var map = JsonSerializer.Deserialize<Dictionary<string, string>>(text.AsSpan(start, end - start + 1))!;

if (args.Length > 0 && args[0] == "list")
{
    Console.WriteLine(map.Count + " icons");
    Console.WriteLine(string.Join(" ", map.Keys.OrderBy(k => k, StringComparer.Ordinal)));
    return;
}

static string Pascal(string kebab) => string.Concat(kebab.Split('-', StringSplitOptions.RemoveEmptyEntries).Select(p => char.ToUpperInvariant(p[0]) + p[1..]));
var wanted = File.ReadAllLines(args[0]).Select(l => l.Trim()).Where(l => l.Length > 0 && !l.StartsWith('#')).Distinct().ToList();
var sb = new StringBuilder();
sb.AppendLine("// <auto-generated> by src/CsCode.Ui/Icons/README.md (Codicons path data, CC-BY-4.0; extracted via MIT-licensed IconPacks.Avalonia.Codicons). Do not edit. </auto-generated>");
sb.AppendLine("namespace CsCode.Ui.Icons;");
sb.AppendLine();
sb.AppendLine("internal static partial class IconPaths");
sb.AppendLine("{");
sb.AppendLine("    /// <summary>Codicon name (kebab-case) to path data in a 100-unit box with y in [-100, 0] (see Icon.cs for the transform).</summary>");
sb.AppendLine("    internal static readonly (string Name, string Data)[] All =");
sb.AppendLine("    [");
var missing = new List<string>();
foreach (var name in wanted)
{
    var key = Pascal(name);
    if (!map.TryGetValue(key, out var data) || data.Length == 0) { missing.Add(name); continue; }
    // A few icons (search, files, debug, terminal...) come in a 150-unit box: scale those into the 100-unit one, or they draw 1.5x too big and
    // are cut off at the top. Their paths use absolute commands only (M L C Q Z), so every number scales.
    var inv = System.Globalization.CultureInfo.InvariantCulture;
    var scale = System.Text.RegularExpressions.Regex.Matches(data, @"-?\d+(\.\d+)?").Any(m => Math.Abs(double.Parse(m.Value, inv)) > 100.5) ? 100.0 / 150 : 1.0;
    // 1 unit = 0.16 px at 16 px, so one decimal is far below visible precision; this shrinks the catalog ~3x.
    data = System.Text.RegularExpressions.Regex.Replace(data, scale == 1 ? @"-?\d+\.\d+" : @"-?\d+(\.\d+)?", m =>
    {
        var s = Math.Round(double.Parse(m.Value, inv) * scale, 1).ToString("0.#", inv);
        return s == "-0" ? "0" : s;
    });
    sb.Append("        (\"").Append(name).Append("\", \"").Append(data.Replace("\\", "\\\\").Replace("\"", "\\\"")).AppendLine("\"),");
}
sb.AppendLine("    ];");
sb.AppendLine("}");
File.WriteAllText(args[1], sb.ToString());
Console.WriteLine($"wrote {wanted.Count - missing.Count} icons; missing: {string.Join(", ", missing)}");
